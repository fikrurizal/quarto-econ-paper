% Title block. Branches on the `author-format` meta key:
%
%   author-format: horizontal  — AEA-style, authors side-by-side
%                                (\and), each author followed by their
%                                affiliation stack. The default for
%                                template.qmd.
%   author-format: numeric     — biomedical-journal style: author names
%                                with superscript affiliation indices
%                                (Smith^{1,2}  Jones^{3}) followed by
%                                a deduplicated numbered affiliation
%                                list. The default for template-numeric.qmd.
%   (anything else, or unset)  — AEA-style, authors stacked vertically,
%                                each with affiliation lines beneath.
%
% The per-value flags `author-format-horizontal` / `author-format-numeric`
% are set by `filters/author-format-flags.lua` (pandoc templates can't
% compare strings directly, so the filter normalises one key into many).
%
% Vendored from the hchulkim/econ-paper-template aea extension; this
% extension is otherwise self-contained.

\def\spacingset#1{\renewcommand{\baselinestretch}%
{#1}\small\normalsize} \spacingset{1}

$if(date)$
\date{$if(articleurl)$\href{$articleurl$}{Link to Latest version}\\ \vspace{1em} $endif$ Last updated $date$}
$endif$

\title{$title$$if(title-footnote)$\thanks{$title-footnote$}$endif$}

$if(author-format-numeric)$
% --- Numeric superscript affiliations -----------------------------------
% Why not authblk? Pandoc emits `\author{Name1 \and Name2 \and Name3}` in
% the preamble before this partial runs. authblk's \author is cumulative,
% so its bookkeeping treats the preamble call as three authors and our
% per-author calls would double them up to six. A single plain-LaTeX
% \author{...} block (which the later call cleanly overwrites) avoids
% the problem.
\author{%
$for(by-author)$
  $it.name.literal$\textsuperscript{$for(it.affiliations)$$it.number$$sep$,$endfor$}$if(it.acknowledgements)$\thanks{$it.acknowledgements$}$endif$$sep$ \quad
$endfor$
  \\[0.6em]
  \normalsize
$for(by-affiliation)$
  \textsuperscript{$it.number$}\,$if(it.department)$$it.department$, $endif$$if(it.name)$$it.name$$endif$$sep$\\
$endfor$
}
$elseif(author-format-horizontal)$
% --- Horizontal AEA-style (authors side-by-side) ------------------------
\author{
$for(by-author)$
$it.name.literal$$if(it.acknowledgements)$\thanks{$it.acknowledgements$}$endif$\\
$for(it.affiliations)$$if(it.department)$$it.department$$if(it.name)$, $endif$$endif$$if(it.name)$$it.name$$endif$$if(it.department)$\\$elseif(it.name)$\\$endif$
$endfor$$sep$\and $endfor$
}
$else$
% --- Vertical AEA-style (authors stacked) -------------------------------
\author{
$for(by-author)$
$it.name.literal$$if(it.acknowledgements)$\thanks{$it.acknowledgements$}$endif$\\
$for(it.affiliations)$$if(it.department)$$it.department$$if(it.name)$, $endif$$endif$$if(it.name)$$it.name$$endif$$if(it.department)$\\$elseif(it.name)$\\$endif$
$endfor$$sep$\\$endfor$
}
$endif$

\maketitle

\bigskip
\bigskip
\begin{abstract}
$abstract$
\end{abstract}

\bigskip
$if(keywords)$
\noindent%
{\it Keywords:} $for(keywords)$$it$$sep$, $endfor$
\vfill
$endif$

\newpage
\spacingset{1.2} % DON'T change the spacing!
