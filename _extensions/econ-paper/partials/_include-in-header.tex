% --- Caption skips -----------------------------------------------------
% belowcaptionskip → space below the caption text (caption-on-top
% layouts; we want appendix table captions hugged to the table body).
% abovecaptionskip → space above the caption text (caption-on-bottom
% figures; 10pt gives a clear image-to-caption gap, 4pt looked glued).
\setlength{\belowcaptionskip}{1pt}
\setlength{\abovecaptionskip}{10pt}

% --- Custom appendix-float caption tightening --------------------------
% Quarto emits each apx* float as
%   \begin{apxatbl}
%   \caption{...}
%   [blank line → \par]
%   \centering{
%   [blank line → \par]
%   {\def\LTcaptype{none}\begin{longtable*}...
% Two \par's contribute ~32pt of vertical air between caption and
% table under \onehalfspacing. \caption's own skip can't see them.
% Patching \caption inside apx envs to add -1em afterwards absorbs
% them; -2em overshoots and overlaps caption with the toprule.
\newcommand{\apxTightCaption}[1]{\apxOrigCaption{#1}\par\vspace*{-1em}}
% Only the TABLE floats get the tight (caption-on-top) patch. Figure floats are
% bottom-captioned, so pulling the following content up overlaps the caption with
% its notes; they use the .fignotes block (below) for spacing instead.
\AtBeginDocument{%
  \AtBeginEnvironment{apxatbl}{\let\apxOrigCaption\caption \let\caption\apxTightCaption}%
  \AtBeginEnvironment{apxbtbl}{\let\apxOrigCaption\caption \let\caption\apxTightCaption}%
}

% --- Force appendix floats to stay at their source position ------------
% Quarto declares the apx* floats with default placement [h], which
% lets LaTeX drift them onto later pages — and a `.tblnotes` block
% underneath then prints before the table it describes. Keep this
% override scoped to appendix custom floats; ordinary main-text floats
% should remain free to move so LaTeX does not leave large blank areas.
\AtBeginDocument{%
  \floatplacement{apxatbl}{H}%
  \floatplacement{apxbtbl}{H}%
  \floatplacement{apxafig}{H}%
  \floatplacement{apxbfig}{H}%
}

% --- Keep notes blocks attached to their appendix float -----------------
% The [H] float mechanism ends with `\box\@currbox \vskip\intextsep`
% (float.sty, \float@endH). That trailing glue is the ONLY legal
% page-break point between the float box and the notes block that
% follows: TeX may break at glue only when the preceding item is a box,
% and every later glue (the tblnotes/fignotes \par and \vspace*s)
% follows glue or a penalty, so it is not breakable. The \nopagebreak[4]
% inside the notes envs therefore executes too late, and a float ending
% near the page bottom ships its ENTIRE notes block to the next page.
% Inserting \penalty\@M between box and glue makes that breakpoint
% illegal: TeX must break BEFORE the float instead, so the float and at
% least the first note lines always share a page, while long notes
% remain internally breakable (tail may overflow) as designed. Scoped to
% the apx* envs via \AtBeginEnvironment -- \patchcmd redefines locally
% within the environment group -- so main-text [H] floats keep normal
% page breaking.
\makeatletter
\newcommand\apxKeepNotesAttached{%
  \patchcmd{\float@endH}
    {\box\@currbox\vskip\intextsep}
    {\box\@currbox\penalty\@M\vskip\intextsep}
    {}{}%
}
\makeatother
\AtBeginDocument{%
  \AtBeginEnvironment{apxatbl}{\apxKeepNotesAttached}%
  \AtBeginEnvironment{apxbtbl}{\apxKeepNotesAttached}%
  \AtBeginEnvironment{apxafig}{\apxKeepNotesAttached}%
  \AtBeginEnvironment{apxbfig}{\apxKeepNotesAttached}%
}

% --- Centre appendix figure captions -----------------------------------
% Appendix figure floats are bottom-captioned; their captions are often
% multi-line, which the caption package justifies (looks left-aligned). Centre
% them. Scoped to the appendix figure float kinds so main-text floats and the
% (top-captioned, tightened) appendix tables are untouched.
\AtBeginDocument{%
  \AtBeginEnvironment{apxafig}{\captionsetup{justification=centering}}%
  \AtBeginEnvironment{apxbfig}{\captionsetup{justification=centering}}%
}

% --- Bibliography entry spacing ----------------------------------------
% Pandoc's CSLReferences env sets `\itemsep = #2\baselineskip` and
% passes #2=1, which leaves a full empty line between references.
% Replace with a small fixed skip. Defined after pandoc's own
% \newenvironment (preamble) so \renewenvironment is safe.
\renewenvironment{CSLReferences}[2]
  {\begin{list}{}{%
    \setlength{\itemindent}{0pt}%
    \setlength{\leftmargin}{0pt}%
    \setlength{\parsep}{0pt}%
    \ifodd #1
      \setlength{\leftmargin}{\cslhangindent}%
      \setlength{\itemindent}{-1\cslhangindent}%
    \fi
    \setlength{\itemsep}{3pt}%
  }}
  {\end{list}}

% --- Clickable in-text citations ---------------------------------------
% pandoc emits each resolved citation as `\citeproc{key}{text}` and
% defines that macro (in `\NewDocumentCommand\citeproc{mm}{...}` in
% the auto-generated preamble) as a natbib bridge: it begins a group,
% binds `\citeproctext` to `text`, then calls `\cite{key}`. With
% natbib loaded, hyperref's `\Hy@cite` patch makes the citation
% clickable.
%
% We INTENTIONALLY don't load natbib here, because pandoc's CSL
% bibliography emits `\bibitem[\citeproctext]{ref-key}` — the
% bibitem label is the literal token `\citeproctext`, which is
% globally empty (only locally bound by `\citeproc{}{}`). natbib's
% `\cite` looks up that label from the .aux at top level (where
% `\citeproctext` is empty) and renders empty citation text. Both
% pandoc issues #9022 / #9031 assume natbib-format bibliographies
% where the label resolves to author-year; the assumption breaks for
% CSL bibliographies (which is exactly what the multibib filter
% produces).
%
% So we override `\citeproc` to bypass `\cite` entirely and jump
% directly to a `\hypertarget` we plant at each `\bibitem`. Keys
% already carry the `ref-` prefix, so no extra prefix is needed.
\RenewDocumentCommand\citeproc{mm}{\hyperlink{#1}{#2}}
\newcommand\econPaperBibitem[2][\null]{%
  \origbibitem[#1]{#2}\hypertarget{#2}{}%
}
\AtBeginEnvironment{CSLReferences}{%
  \let\origbibitem\bibitem%
  \let\bibitem\econPaperBibitem%
}

% --- Journal-style notes block -----------------------------------------
% Triggered by `::: {.tblnotes} ... :::` / `::: {.fignotes} ... :::` in
% markdown; filters/div-to-env.lua wraps the div in the matching env.
%
% tblnotes (below a TOP-captioned table) and fignotes (below a BOTTOM-captioned
% figure) are kept visually identical: same font (\footnotesize), no paragraph
% indent, the SAME gap above the notes (\noteTopGap) and the SAME gap below
% (\noteBotGap). They differ only in how much post-float skip they must cancel
% to land that common gap, because a table float has TWO stacked floats above
% it (the apx* env + the inner \begin{table}[H] from HOLD_position) while a
% figure float has ONE. `\setlength{\parindent}{0pt}` zeros the first-line
% indent on every paragraph (a bare \noindent only fixes the first and is
% swallowed by the \par the surrounding blank lines insert).
%
% \noteTopGap : visible space between the table/figure and its notes.
% \noteBotGap : space between the notes and the following paragraph/section.
\newlength{\noteTopGap}\setlength{\noteTopGap}{6pt}
\newlength{\noteBotGap}\setlength{\noteBotGap}{1em}

% Both note blocks flow as normal (breakable) text rather than an unbreakable
% minipage: the note stays attached to its float (first lines on the float's
% page, tail overflowing) instead of the whole block jumping to the next page
% when a tall table/figure leaves little room. `\nopagebreak[4]` discourages a
% break immediately after the float. tblnotes and fignotes are identical except
% for how much post-float skip they cancel (a table float has two stacked floats
% above it, a figure one).
\NewEnviron{tblnotes}{%
  \par\vspace*{-2\intextsep}\vspace*{\noteTopGap}%
  \nopagebreak[4]%
  \noindent{%
    \footnotesize%
    \setlength{\parindent}{0pt}%
    \BODY\par
  }%
  \vspace{\noteBotGap}%
}

\NewEnviron{fignotes}{%
  \par\vspace*{-1\intextsep}\vspace*{\noteTopGap}%
  \nopagebreak[4]%
  \noindent{%
    \footnotesize%
    \setlength{\parindent}{0pt}%
    \BODY\par
  }%
  \vspace{\noteBotGap}%
}
